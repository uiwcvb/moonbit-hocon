import assert from 'node:assert/strict';
import {pathToFileURL} from 'node:url';
const [base,current]=await Promise.all(process.argv.slice(2).map(s=>import(pathToFileURL(s))));
const median=xs=>xs.sort((a,b)=>a-b)[Math.floor(xs.length/2)];
for(const [name,items] of [['unicode',['１２３４５','-１２３４５','١٢٣٤٥','१२३४५']],['mixed',['1٢3','123٤5','12３45','-1234٥']],['ascii',['12345','-12345','123456789','0']]]){
 const source='values='+JSON.stringify(Array.from({length:32},(_,i)=>items[i%items.length]));
 const configs=[base.Config.load(source),current.Config.load(source)];assert.deepEqual(configs[0].getDoubleList('values'),configs[1].getDoubleList('values'));
 const samples=[[],[]];for(const config of configs)for(let i=0;i<500;i++)config.getDoubleList('values');
 for(let pass=0;pass<10;pass++)for(const index of pass%2?[1,0]:[0,1]){const start=performance.now();for(let i=0;i<300;i++)configs[index].getDoubleList('values');samples[index].push((performance.now()-start)/300);}
 console.log(JSON.stringify({name,baseMs:median(samples[0]),currentMs:median(samples[1]),ratio:median(samples[1])/median(samples[0])}));
}
